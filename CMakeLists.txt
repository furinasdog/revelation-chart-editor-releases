cmake_minimum_required(VERSION 3.20)
project(RevelationEditor LANGUAGES CXX)

set(GODOT_CPP_DIR "" CACHE PATH "godot-cpp checkout targeting Godot 4.5")
set(ONNXRUNTIME_DIR "" CACHE PATH "ONNX Runtime C/C++ SDK")
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/native/build_profile.json" CACHE FILEPATH "Godot build profile")
set(GODOTCPP_TARGET "template_release" CACHE STRING "Godot binding target")
set(GODOTCPP_DISABLE_EXCEPTIONS OFF CACHE BOOL "Allow C++ exceptions")
add_subdirectory("${GODOT_CPP_DIR}" godot-cpp)

option(REVELATION_BUILD_AUDIO_ENGINES "Build the bundled madmom, Essentia and RMVPE executables" OFF)
set(REVELATION_AUDIO_NATIVE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/revelation-audio-native" CACHE PATH "C++ audio engines source checkout")
if(REVELATION_BUILD_AUDIO_ENGINES)
  set(ONNXRUNTIME_ROOT "${ONNXRUNTIME_DIR}" CACHE PATH "ONNX Runtime C++ SDK for the audio engines")
  set(REVELATION_EDITOR_BIN "${CMAKE_CURRENT_SOURCE_DIR}/editor/bin" CACHE PATH "Editor runtime binaries")
  add_subdirectory("${REVELATION_AUDIO_NATIVE_DIR}" audio-engines)
  # Model installation belongs to the Hub. The editor's native worker reads verified local files.
  set_source_files_properties("${REVELATION_AUDIO_NATIVE_DIR}/src/model_files_windows.cpp"
    TARGET_DIRECTORY revelation-madmom PROPERTIES HEADER_FILE_ONLY ON)
  target_sources(revelation-madmom PRIVATE native/madmom_local_models.cpp)
endif()

option(REVELATION_BUILD_MUSIC2EMO "Build the native Music2Emo emotion analyzer" OFF)
set(REVELATION_MUSIC2EMO_DIR "${CMAKE_CURRENT_SOURCE_DIR}/music2emo-native" CACHE PATH "Music2Emo C++ source checkout")
if(REVELATION_BUILD_MUSIC2EMO)
  set(ONNXRUNTIME_ROOT "${ONNXRUNTIME_DIR}" CACHE PATH "ONNX Runtime C++ SDK for Music2Emo")
  set(REVELATION_EDITOR_BIN "${CMAKE_CURRENT_SOURCE_DIR}/editor/bin" CACHE PATH "Editor runtime binaries")
  add_subdirectory("${REVELATION_MUSIC2EMO_DIR}" music2emo)
endif()

option(REVELATION_BUILD_HTDEMUCS "Build the native HTDemucs four-stem separator" OFF)
set(REVELATION_HTDEMUCS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/revelation-htdemucs-native" CACHE PATH "HTDemucs C++ source checkout")
if(REVELATION_BUILD_HTDEMUCS)
  add_subdirectory("${REVELATION_HTDEMUCS_DIR}" htdemucs)
  set_target_properties(revelation-stems PROPERTIES RUNTIME_OUTPUT_DIRECTORY "$<1:${CMAKE_CURRENT_SOURCE_DIR}/editor/bin>")
endif()

find_path(ORT_INCLUDE onnxruntime_cxx_api.h PATHS "${ONNXRUNTIME_DIR}/include" REQUIRED NO_DEFAULT_PATH)
find_library(ORT_LIBRARY onnxruntime PATHS "${ONNXRUNTIME_DIR}/lib" REQUIRED NO_DEFAULT_PATH)
add_library(revelation_rhythm SHARED native/rhythm.cpp)
target_compile_features(revelation_rhythm PRIVATE cxx_std_17)
target_include_directories(revelation_rhythm PRIVATE "${ORT_INCLUDE}")
target_link_libraries(revelation_rhythm PRIVATE godot-cpp "${ORT_LIBRARY}")
if(MSVC)
  target_compile_options(revelation_rhythm PRIVATE /EHsc /utf-8)
endif()

if(WIN32)
  set(EDITOR_BIN "${CMAKE_CURRENT_SOURCE_DIR}/editor/bin")
  set_target_properties(revelation_rhythm PROPERTIES RUNTIME_OUTPUT_DIRECTORY "$<1:${EDITOR_BIN}>")
  find_file(ORT_DLL onnxruntime.dll PATHS "${ONNXRUNTIME_DIR}/lib" "${ONNXRUNTIME_DIR}/bin" REQUIRED NO_DEFAULT_PATH)
  add_custom_command(TARGET revelation_rhythm POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${ORT_DLL}" "${EDITOR_BIN}/onnxruntime.dll")
endif()
